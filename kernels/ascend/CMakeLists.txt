cmake_minimum_required(VERSION 3.19)
project(plait_w4a16_ascend LANGUAGES C CXX)

set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit")
set(SOC_VERSION "$ENV{ASCEND_SOC_VERSION}" CACHE STRING "CANN kernel target")
option(PLAIT_ASCEND_HUMMING "Build the dense Humming Cube kernel on Ascend 950" OFF)
if(NOT SOC_VERSION)
  message(FATAL_ERROR "Pass -DSOC_VERSION for the installed Ascend target")
endif()
set(RUN_MODE npu)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(cann_arch "${ASCEND_CANN_PACKAGE_PATH}/${CMAKE_SYSTEM_PROCESSOR}-linux")
include("${cann_arch}/tikcpp/ascendc_kernel_cmake/ascendc.cmake")

set(kernel_sources w4a16_kernel.cpp)
if(SOC_VERSION MATCHES "^Ascend950")
  set(enable_integer_cube ON)
  list(APPEND kernel_sources w4_integer_kernel.cpp w4_integer_cube.cpp w4_quantize_kernel.cpp w4_marlin_kernel.cpp)
endif()
if(PLAIT_ASCEND_HUMMING)
  if(NOT enable_integer_cube)
    message(FATAL_ERROR "PLAIT_ASCEND_HUMMING requires an Ascend 950 target")
  endif()
  list(APPEND kernel_sources w4_humming_kernel.cpp w4_humming_cube.cpp)
endif()
ascendc_library(plait_ascend_kernel STATIC ${kernel_sources})
add_library(plait-w4a16-ascend SHARED w4a16_host.c)
if(enable_integer_cube)
  target_compile_definitions(plait-w4a16-ascend PRIVATE PLAIT_ASCEND_INTEGER_CUBE)
endif()
if(PLAIT_ASCEND_HUMMING)
  target_compile_definitions(plait-w4a16-ascend PRIVATE PLAIT_ASCEND_HUMMING)
endif()
target_include_directories(plait-w4a16-ascend PRIVATE "${cann_arch}/include")
target_link_directories(plait-w4a16-ascend PRIVATE "${cann_arch}/lib64")
target_link_libraries(plait-w4a16-ascend PRIVATE plait_ascend_kernel ascendcl pthread)
